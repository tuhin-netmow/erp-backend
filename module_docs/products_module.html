// File: src/modules/products/products.controller.js:8-30
async getAllProducts(req, res) {
    try {
        const page = parseInt(req.query.page) || 1;
        const limit = parseInt(req.query.limit) || 10;
        const filters = {
            category_id: req.query.category_id,
            is_active: req.query.is_active !== 'false',
            stock_status: req.query.stock_status,
            search: req.query.search
        };

        const offset = (page - 1) * limit;
        const result = await ProductRepository.findAll(filters, limit, offset);

        return successWithPagination(res, 'Products retrieved successfully', result.rows, {
            total: result.count,
            page,
            limit
        });
    } catch (err) {
        return error(res, err.message, 500);
    }
}

// File: src/modules/products/products.repository.js:5-59
async findAll(filters = {}, limit = 10, offset = 0) {
    const where = {};

    if (filters.category_id) where.category_id = filters.category_id;
    if (filters.is_active !== undefined) where.is_active = filters.is_active;

    if (filters.stock_status) {
        if (filters.stock_status === 'available') {
            where.stock_quantity = { [Op.gt]: 0 };
        } else if (filters.stock_status === 'out_of_stock') {
            where.stock_quantity = 0;
        } else if (filters.stock_status === 'low_stock') {
            where.stock_quantity = {
                [Op.and]: [
                    { [Op.lte]: Sequelize.col('min_stock_level') },
                    { [Op.gt]: 0 }
                ]
            };
        }
    }

    if (filters.search) {
        where[Op.or] = [
            { name: { [Op.like]: `%${filters.search}%` } },
            { sku: { [Op.like]: `%${filters.search}%` } },
            { barcode: { [Op.like]: `%${filters.search}%` } }
        ];
    }

    return await Product.findAndCountAll({
        where,
        include: [
            { model: Category, as: 'category' },
            { model: Unit, as: 'unit' },
            { model: ProductImage, as: 'images',
              order: [['sort_order', 'ASC'], ['is_primary', 'DESC']] }
        ],
        limit, offset,
        order: [['created_at', 'DESC']],
        distinct: true
    });
}

// File: src/modules/products/products.service.js:6-42
async create(data, userId = null) {
    const { ProductRepository } = require('../products/products.repository');
    const { generateNumber } = require('../../core/utils/numberGenerator');

    // Auto-generate SKU if not provided
    if (!data.sku) {
        let sku;
        let isUnique = false;
        let attempts = 0;

        while (!isUnique && attempts < 10) {
            sku = generateNumber('product'); // e.g., PRD-1738400000000
            const existing = await ProductRepository.findBySku(sku);
            if (!existing) isUnique = true;
            attempts++;
        }

        if (!isUnique) {
            throw new Error('Could not generate unique SKU after 10 attempts');
        }
        data.sku = sku;
    } else {
        // Check uniqueness if SKU provided
        const existing = await ProductRepository.findBySku(data.sku);
        if (existing) {
            throw new Error('SKU already exists');
        }
    }

    // Set created_by
    data.created_by = userId;

    // Create product
    const product = await ProductRepository.create(data);

    // Handle gallery_items
    if (data.gallery_items && Array.isArray(data.gallery_items) && data.gallery_items.length > 0) {
        const images = data.gallery_items.map((url, index) => ({
            product_id: product.id,
            image_url: url,
            is_primary: index === 0 && !data.image_url, // First image is primary if no image_url
            sort_order: index,
            caption: null
        }));

        await ProductImageRepository.createMultiple(images);
    }

    // Record initial stock movement if applicable
    const initialStock = data.initial_stock || data.stock_quantity || 0;
    if (initialStock > 0) {
        await this._recordStockMovement(product.id, 'adjustment', initialStock, 'Initial stock', userId);
    }

    // Fetch full product with associations
    const fullProduct = await ProductRepository.findById(product.id);
    return this._transformProduct(fullProduct);
}

// File: src/modules/products/products.service.js:132-160
async updateStock(productId, operation, quantity, notes = null, userId = null) {
    const { ProductRepository } = require('../products/products.repository');

    const product = await ProductRepository.findById(productId);
    if (!product) {
        throw new Error('Product not found');
    }

    let newQuantity = product.stock_quantity;
    let movementQuantity = quantity;

    switch (operation) {
        case 'add':
            newQuantity += quantity;
            break;
        case 'subtract':
            newQuantity = Math.max(0, newQuantity - quantity);
            movementQuantity = -quantity;
            break;
        case 'set':
            movementQuantity = quantity - newQuantity;
            newQuantity = quantity;
            break;
        default:
            throw new Error('Invalid operation. Use: add, subtract, or set');
    }

    await ProductRepository.update(productId, {
        stock_quantity: newQuantity
    });

    await this._recordStockMovement(
        productId,
        'adjustment',
        movementQuantity,
        notes || `Stock ${operation}: ${quantity}`,
        userId
    );

    const updated = await ProductRepository.findById(productId);
    return this._transformProduct(updated);
}

async getStats() {
    const [activeProducts, totalProducts, lowStockProducts,
           outOfStockProducts, financialResults] = await Promise.all([
        Product.count({ where: { is_active: true } }),
        Product.count(),
        Product.count({
            where: {
                stock_quantity: {
                    [Op.and]: [
                        { [Op.lte]: Sequelize.col('min_stock_level') },
                        { [Op.gt]: 0 }
                    ]
                },
                is_active: true
            }
        }),
        Product.count({
            where: {
                stock_quantity: 0,
                is_active: true
            }
        }),
        Product.findOne({
            attributes: [
                [Sequelize.fn('SUM', Sequelize.literal('stock_quantity * COALESCE(cost, 0)')), 'totalCost'],
                [Sequelize.fn('SUM', Sequelize.literal('stock_quantity * COALESCE(price, 0)')), 'totalPrice']
            ],
            where: { is_active: true },
            raw: true
        })
    ]);

    const totalCost = parseFloat(financialResults?.totalCost || 0);
    const totalPrice = parseFloat(financialResults?.totalPrice || 0);

    return {
        activeProducts,
        totalProducts,
        lowStockProducts,
        outOfStockProducts,
        totalCost,
        totalPrice,
        totalProfit: totalPrice - totalCost
    };
}

// File: src/modules/products/products.repository.js:218-267
async getProductOrders(productId, filters = {}, limit = 10, offset = 0) {
    const { Order, OrderItem } = require('../sales/sales.models');
    const { Customer } = require('../customers/customers.model');

    const where = {};

    if (filters.status) where.status = filters.status;

    if (filters.start_date || filters.end_date) {
        where.order_date = {};
        if (filters.start_date) {
            where.order_date[Op.gte] = new Date(filters.start_date);
        }
        if (filters.end_date) {
            where.order_date[Op.lte] = new Date(filters.end_date);
        }
    }

    return await Order.findAndCountAll({
        where,
        include: [
            {
                model: OrderItem,
                as: 'items',
                where: { product_id: productId },
                required: true,
                include: [
                    {
                        model: Product,
                        as: 'product',
                        attributes: ['id', 'name', 'sku', 'image_url']
                    }
                ]
            },
            {
                model: Customer,
                as: 'customer',
                attributes: ['id', 'name', 'email', 'phone', 'company']
            }
        ],
        limit,
        offset,
        order: [['order_date', 'DESC']],
        distinct: true
    });
}
