async createManualJournal(data) {
    // 1. Validate Debit = Credit
    const totalDebit = data.entries.reduce((sum, e) => sum + e.debit, 0);
    const totalCredit = data.entries.reduce((sum, e) => sum + e.credit, 0);

    if (Math.abs(totalDebit - totalCredit) > 0.01) {
        throw new Error(`Debit (${totalDebit}) must equal Credit (${totalCredit})`);
    }

    // 2. Use repository to create journal with lines
    const journalData = {
        date: data.date,
        narration: data.narration || 'Manual Journal Entry',
        reference_type: 'MANUAL',
        reference_id: null
    };

    return await sequelize.transaction(async (t) => {
        return await AccountingRepository.createJournalEntry(journalData, data.entries, t);
    });
}

async createJournalEntry(journalData, linesData, transaction) {
    // 1. Create Journal Header
    const journal = await Journal.create(journalData, { transaction });

    // 2. Create Journal Lines
    const lines = linesData.map(line => ({
        ...line,
        journal_id: journal.id
    }));

    await JournalLine.bulkCreate(lines, { transaction });

    return journal;
}

async processTransaction(data, existingTransaction = null) {
    return await sequelize.transaction(async (t) => {
        // 1. Create the Single Entry Transaction
        const transaction = await AccountingRepository.createTransaction(data, t);

        // 2. Build Journal Entries based on type
        const amount = parseFloat(data.amount);
        const tax_amount = parseFloat(data.tax_amount || 0);
        const linesData = [];

        switch (data.type) {
            case 'SALES': {
                // Dr Cash/Bank/AR (Full Amount)
                const drCode = (data.payment_mode === 'CASH') ? ACCOUNTS.CASH :
                    (data.payment_mode === 'BANK') ? ACCOUNTS.BANK : ACCOUNTS.AR;
                linesData.push({
                    account_id: await this.getAccountId(drCode),
                    debit: amount,
                    credit: 0
                });

                // Cr Sales (Amount - Tax)
                const subtotal = amount - tax_amount;
                linesData.push({
                    account_id: await this.getAccountId(ACCOUNTS.SALES),
                    debit: 0,
                    credit: subtotal
                });

                // Cr Tax Payable (Tax)
                if (tax_amount > 0) {
                    linesData.push({
                        account_id: await this.getAccountId(ACCOUNTS.TAX_PAYABLE),
                        debit: 0,
                        credit: tax_amount
                    });
                }
                break;
            }
            // ... other types
        }

        // 3. Create Journal Entry
        const journalData = {
            date: data.date || new Date(),
            reference_type: 'TRANSACTION',
            reference_id: transaction.id,
            narration: data.description || `${data.type} Transaction`
        };

        const journal = await AccountingRepository.createJournalEntry(journalData, linesData, t);

        return { transaction, journal };
    });
}

async findAllTransactions(filters = {}, limit = 10, offset = 0) {
    const where = {};
    if (filters.type) where.type = filters.type;

    // Date aliases
    const fromDate = filters.from || filters.start_date;
    const toDate = filters.to || filters.end_date;
    const exactDate = filters.date;

    if (fromDate && toDate) {
        where.date = { [Op.between]: [fromDate, toDate] };
    } else if (exactDate) {
        where.date = exactDate;
    }

    return await Transaction.findAndCountAll({
        where,
        limit,
        offset,
        order: [['date', 'DESC'], ['id', 'DESC']]
    });
}

async getLedgerReport(accountId, filters) {
    // Get opening balance and lines from repository
    const { openingBalance, lines } = await AccountingRepository.getLedgerReport(accountId, filters);

    // Get account details
    const account = await AccountingRepository.findAccountById(accountId);
    if (!account) throw new Error('Account not found');

    const isDebitNature = ['ASSET', 'EXPENSE'].includes(account.type);
    let currentBalance = openingBalance;

    // Add running balance to each line
    const enhancedLines = lines.map(line => {
        const dr = parseFloat(line.debit);
        const cr = parseFloat(line.credit);

        if (isDebitNature) {
            currentBalance += (dr - cr);
        } else {
            currentBalance += (cr - dr);
        }

        return {
            date: line.Journal.date,
            narration: line.Journal.narration,
            debit: dr,
            credit: cr,
            balance: currentBalance,

async getLedgerReport(accountId, filters = {}) {
    let openingBalance = 0;

    // Calculate opening balance if 'from' date is provided
    if (filters.from) {
        const prevLines = await JournalLine.findAll({
            where: { account_id: accountId },
            include: [{
                model: Journal,
                where: {
                    date: { [Op.lt]: filters.from }
                }
            }]
        });

        const account = await Account.findByPk(accountId);
        const isDebitNature = ['ASSET', 'EXPENSE'].includes(account.type);

        prevLines.forEach(line => {
            const debit = parseFloat(line.debit) || 0;
            const credit = parseFloat(line.credit) || 0;
            if (isDebitNature) {
                openingBalance += (debit - credit);
            } else {
                openingBalance += (credit - debit);
            }
        });
    }

    // Get lines for the period
    const where = { account_id: accountId };
    const journalWhere = {};
    if (filters.from && filters.to) {
        journalWhere.date = { [Op.between]: [filters.from, filters.to] };
    }

    const lines = await JournalLine.findAll({
        where,
        include: [{
            model: Journal,
            where: journalWhere
        }],
        order: [[{ model: Journal }, 'date', 'ASC']]
    });

    return { openingBalance, lines };
}

async getProfitAndLoss(filters = {}) {
    const incomeAccounts = await AccountingRepository.findAccountsByTypes(['INCOME']);
    const expenseAccounts = await AccountingRepository.findAccountsByTypes(['EXPENSE']);

    const getAccountBalances = async (accounts) => {
        const results = [];
        for (const acc of accounts) {
            const lines = await AccountingRepository.getAccountLinesForPeriod(acc.id, filters);
            const dr = lines.reduce((sum, l) => sum + (parseFloat(l.debit) || 0), 0);
            const cr = lines.reduce((sum, l) => sum + (parseFloat(l.credit) || 0), 0);

            // Income: Net = Credit - Debit
            // Expense: Net = Debit - Credit
            const net = (acc.type === 'INCOME') ? (cr - dr) : (dr - cr);

            if (net !== 0) {
                results.push({
                    code: acc.code,
                    name: acc.name,
                    amount: Math.abs(net)
                });
            }
        }
        return results;
    };

    const income = await getAccountBalances(incomeAccounts);
    const expense = await getAccountBalances(expenseAccounts);

    const totalIncome = income.reduce((sum, i) => sum + i.amount, 0);
    const totalExpense = expense.reduce((sum, e) => sum + e.amount, 0);
    const netProfit = totalIncome - totalExpense;

    return {
        income,
        expense,
        total_income: totalIncome,
        total_expense: totalExpense,
        net_profit: netProfit
    };
}

async getBalanceSheet(date) {
    // Get P&L for retained earnings
    const pAndL = await this.getProfitAndLoss({ to: date });
    const retainedEarnings = pAndL.net_profit;

    // Get Assets
    const assetAccounts = await AccountingRepository.findAccountsByTypes(['ASSET']);
    const assets = await this.getAccountBalances(assetAccounts, date);

    // Get Liabilities
    const liabilityAccounts = await AccountingRepository.findAccountsByTypes(['LIABILITY']);
    const liabilities = await this.getAccountBalances(liabilityAccounts, date);

    // Get Equity
    const equityAccounts = await AccountingRepository.findAccountsByTypes(['EQUITY']);
    const equities = await this.getAccountBalances(equityAccounts, date);

    const totalAssets = assets.reduce((sum, a) => sum + a.balance, 0);
    const totalLiabilities = liabilities.reduce((sum, l) => sum + l.balance, 0);
    const totalEquity = equities.reduce((sum, e) => sum + e.balance, 0) + retainedEarnings;

    return {
        assets,
        liabilities,
        equity: [
            ...equities,
            { name: 'Retained Earnings', balance: retainedEarnings }
        ],
        total_assets: totalAssets,
        total_liabilities: totalLiabilities,
        total_equity: totalEquity
    };
}
