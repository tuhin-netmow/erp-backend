Automatic Calculations: The system automatically calculates total_amount, tax_amount (based on product purchase tax), discount_amount, and grand_total from the items provided.
Inventory Side Effects: Creating a purchase return automatically:
  • Decreases products.stock_quantity for returned items
  • Creates stock_movements records for inventory tracking
  • Associates return with original PO if purchase_order_id provided

async createPurchaseReturn(data, userId) {
    let { items } = data;
    let returnInfo = { ...data };

    // Validate items
    if (!Array.isArray(items) || items.length === 0) {
        throw new Error('Purchase return must contain at least one item');
    }

    // Generate return number
    const return_number = `PRRTN-${Date.now()}-${Math.random().toString(36).substring(2, 8).toUpperCase()}`;

    // Fetch products and validate
    const productIds = items.map(i => i.product_id);
    const products = await ProductRepository.findByIds(productIds);
    const productMap = new Map(products.map(p => [String(p.id), p]));

    // Calculate totals
    let total_amount = 0;
    let discount_amount = 0;
    let tax_amount = 0;

    const processedItems = items.map(item => {
        const quantity = Number(item.quantity) || 0;
        const unit_cost = Number(item.unit_cost) || 0;
        const discount = Number(item.discount) || 0;

        if (quantity <= 0 || unit_cost < 0) {
            throw new Error('Invalid quantity or unit cost');
        }

        const product = productMap.get(String(item.product_id));
        if (!product) {
            throw new Error(`Product not found: ${item.product_id}`);
        }

        const tax_percent = Number(product.purchase_tax) || 0;
        const line_subtotal = quantity * unit_cost;
        const line_total = Math.max(line_subtotal - discount, 0);
        const item_tax_amount = (line_total * tax_percent) / 100;

        discount_amount += discount;
        total_amount += line_total;
        tax_amount += item_tax_amount;

        return {
            ...item,
            quantity,
            unit_cost,
            discount,
            line_total,
            tax_amount: item_tax_amount
        };
    });

    const grand_total = total_amount - discount_amount + tax_amount;

    // Create purchase return
    const returnData = {
        ...returnInfo,
        return_number,
        total_amount,
        discount_amount,
        tax_amount,
        grand_total,
        created_by: userId
    };

    const purchaseReturn = await PurchaseReturnRepository.create(returnData, processedItems);

    // Update stock quantities
    for (const item of processedItems) {
        await ProductRepository.decreaseStock(item.product_id, item.quantity);
    }

    // Create stock movements
    for (const item of processedItems) {
        await StockMovementRepository.create({
            product_id: item.product_id,
            quantity: -item.quantity,
            movement_type: 'purchase_return',
            reference_id: purchaseReturn.id,
            reference_number: return_number
        });
    }

    return purchaseReturn;
}

async create(data, items) {
    const transaction = await sequelize.transaction();

    try {
        // Create purchase return
        const purchaseReturn = await PurchaseReturn.create(data, { transaction });

        // Create return items
        const itemsWithReturnId = items.map(item => ({
            ...item,
            purchase_return_id: purchaseReturn.id
        }));

        await PurchaseReturnItem.bulkCreate(itemsWithReturnId, { transaction });

        await transaction.commit();
        return purchaseReturn;
    } catch (error) {
        await transaction.rollback();
        throw error;
    }
}

async getAllPurchaseReturns(filters = {}, page = 1, limit = 10) {
    const offset = (page - 1) * limit;

    const result = await PurchaseReturnRepository.findAll(filters, limit, offset);

    // Calculate totals for each return
    const data = result.rows.map(row => {
        const pr = row.get({ plain: true });
        const totalAmount = parseFloat(pr.total_amount || 0);
        const discountAmount = parseFloat(pr.discount_amount || 0);
        const taxAmount = parseFloat(pr.tax_amount || 0);
        const refunded = parseFloat(pr.total_refunded_amount || 0);

        const grandTotal = totalAmount - discountAmount + taxAmount;
        const dueRefund = grandTotal - refunded;

        return {
            ...pr,
            grand_total: grandTotal,
            total_refunded_amount: refunded,
            due_refund_amount: dueRefund > 0 ? dueRefund : 0
        };
    });

    return {
        data,
        total: result.count
    };
}

async findAll(filters = {}, limit = 10, offset = 0) {
    const where = {};

    // Apply filters
    if (filters.status) {
        where.status = filters.status;
    }

    if (filters.supplier_id) {
        where.supplier_id = filters.supplier_id;
    }

    if (filters.search) {
        where.return_number = {
            [Op.like]: `%${filters.search}%`
        };
    }

    const { count, rows } = await PurchaseReturn.findAndCountAll({
        where,
        include: [
            {
                model: Supplier,
                as: 'supplier',
                attributes: ['id', 'name']
            },
            {
                model: PurchaseReturnItem,
                as: 'items',
                include: [
                    {
                        model: Product,
                        as: 'product',
                        attributes: ['id', 'name', 'sku']
                    }
                ]
            }
        ],
        attributes: {
            include: [
                [
                    sequelize.literal(`(
                        SELECT COALESCE(SUM(amount), 0)
                        FROM purchase_return_payments
                        WHERE purchase_return_id = PurchaseReturn.id
                        AND status = 'completed'
                    )`),
                    'total_refunded_amount'
                ]
            ]
        },
        limit,
        offset,
        order: [['created_at', 'DESC']]
    });

    return { count, rows };
}

SQL Equivalent:
SELECT
    pr.*,
    (
        SELECT COALESCE(SUM(amount), 0)
        FROM purchase_return_payments
        WHERE purchase_return_id = pr.id AND status = 'completed'
    ) AS total_refunded_amount,
    s.id AS supplier.id, s.name AS supplier.name
FROM purchase_returns AS pr
LEFT JOIN suppliers AS s ON s.id = pr.supplier_id
LEFT JOIN purchase_return_items AS pri ON pri.purchase_return_id = pr.id
WHERE pr.status = ?
ORDER BY pr.created_at DESC
LIMIT ? OFFSET ?;

📌 DATABASE SCHEMA

Database Tables

purchase_returns

Main purchase returns table.

Column Type Description
id INT (PK) Primary key, auto-increment
return_number VARCHAR(255) Unique return number (auto-generated)
supplier_id INT (FK) Foreign key to suppliers
purchase_order_id INT (FK) Original purchase order ID (optional)
return_date DATE Return date
status ENUM Return status (pending, approved, returned, cancelled)
total_amount DECIMAL Subtotal before tax and discount
tax_amount DECIMAL Tax amount
discount_amount DECIMAL Discount amount
grand_total DECIMAL Total refund amount
payment_status ENUM Payment status (pending, partial, paid)
notes TEXT Return notes
created_by INT (FK) User who created the return
created_at TIMESTAMP Creation timestamp
updated_at TIMESTAMP Last update timestamp

purchase_return_items

Line items for purchase returns.

Column Type Description
id INT (PK) Primary key
purchase_return_id INT (FK) Foreign key to purchase_returns
product_id INT (FK) Foreign key to products
quantity INT Quantity returned
unit_cost DECIMAL Unit cost for refund
discount DECIMAL Discount per item
line_total DECIMAL Line total (quantity × cost - discount)

purchase_return_invoices

Debit notes issued for purchase returns.

Column Type Description
id INT (PK) Primary key
invoice_number VARCHAR(255) Debit note number (auto-generated)
purchase_return_id INT (FK) Foreign key to purchase_returns
total_amount DECIMAL Debit note amount
status ENUM Status (issued, credited, cancelled)
invoice_date DATE Debit note date

purchase_return_payments

Refund payments received from suppliers.

Column Type Description
id INT (PK) Primary key
purchase_return_id INT (FK) Foreign key to purchase_returns
invoice_id INT (FK) Associated invoice (optional)
amount DECIMAL Refund amount received
payment_method VARCHAR(50) Payment method (CASH, BANK_TRANSFER, CHECK, etc.)
reference_number VARCHAR(255) Transaction reference
payment_date DATE Payment date
status ENUM Status (pending, completed, failed)
