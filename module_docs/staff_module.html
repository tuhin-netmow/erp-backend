async getAllStaffs(req, res) {
    try {
        const page = parseInt(req.query.page) || 1;
        const limit = parseInt(req.query.limit) || 10;
        const filters = {
            department_id: req.query.department_id,
            role_id: req.query.role_id,
            status: req.query.status,
            search: req.query.search
        };

        const result = await StaffService.getAllStaffs(filters, page, limit);
        return successWithPagination(res, 'Staff members retrieved successfully', result.data, {
            total: result.total,
            page,
            limit
        });
    } catch (err) {
        return error(res, err.message, 500);
    }
}

async getAllStaffs(filters = {}, page = 1, limit = 10) {
    const offset = (page - 1) * limit;

    return {
        data: result.rows,
        total: result.count
    };
}

async findAll(filters = {}, limit = 10, offset = 0) {
    const where = {};

    // Department filter
    if (filters.department_id) {
        where.department_id = filters.department_id;
    }

    // Role filter
    if (filters.role_id) {
        where.role_id = filters.role_id;
    }

    // Status filter
    if (filters.status) {
        where.status = filters.status;
    }

    // Search across multiple fields
    if (filters.search) {
        where[Op.or] = [
            { first_name: { [Op.like]: `%${filters.search}%` } },
            { last_name: { [Op.like]: `%${filters.search}%` } },
            { name: { [Op.like]: `%${filters.search}%` } },
            { email: { [Op.like]: `%${filters.search}%` } },
            { position: { [Op.like]: `%${filters.search}%` } }
        ];
    }

    return await Staff.findAndCountAll({
        where,
        attributes: {
            include: [
                [
                    sequelize.literal(`(
                        SELECT COALESCE(SUM(amount), 0)
                        FROM payroll_advances
                        WHERE payroll_advances.staff_id = User.id
                    )`),
                    'total_advance'
                ],
                [
                    sequelize.literal(`(
                        SELECT COALESCE(SUM(returned_amount), 0)
                        FROM payroll_advances
                        WHERE payroll_advances.staff_id = User.id
                    )`),
                    'total_returned'
                ]
            ]
        },
        limit,
        offset,
        order: [['created_at', 'DESC']],
        include: [
            {
                model: Department,
                as: 'department',
                attributes: ['id', 'name', 'description']
            },
            {
                model: Role,
                as: 'role',
                attributes: ['id', 'name', 'display_name']
            },
            {
                model: PayrollStructure,
                as: 'payrollStructure',
                attributes: ['basic_salary', 'allowances', 'deductions', 'bank_details'],
                required: false // LEFT JOIN
            }
        ]
    });
}

async createStaff(data, userId) {
    // Check email uniqueness
    if (data.email) {
        const existing = await StaffRepository.findByEmail(data.email);
        if (existing) {
            throw new Error('Staff with this email already exists');
        }
    }

    // Hash password if provided, or generate default
    if (data.password) {
        data.password = await bcrypt.hash(data.password, 10);
    } else {
        // Generate a random password if not provided
        const randomPass = Math.random().toString(36).slice(-8);
        data.password = await bcrypt.hash(randomPass, 10);
    }

    // Sanitize data: convert empty strings to null
    const cleanData = { ...data };
    ['hire_date', 'thumb_url', 'position', 'phone', 'address', 'city', 'state', 'country', 'postal_code', 'notes'].forEach(field => {
        if (cleanData[field] === '') {
            cleanData[field] = null;
        }
    });

    return await StaffRepository.create({ ...cleanData, created_by: userId });
}

    const result = await require('../sales/salesroute.repository').getStaffRoutes(filters, limit, (page - 1) * limit);

    const formattedData = result.rows.map(staff => {
        const routes = staff.assignedRoutes || [];

        // Calculate aggregations
        let completedOrdersCount = 0;

        const formattedRoutes = routes.map(route => {
            const customers = route.customers || [];
            let routeOrderCount = 0;
            let routeCompletedOrders = 0;

            customers.forEach(customer => {
                const orders = customer.orders || [];
                routeOrderCount += orders.length;

                const completed = orders.filter(o => ['delivered', 'completed', 'shipped'].includes(o.status)).length;
                routeCompletedOrders += completed;
            });

            completedOrdersCount += routeCompletedOrders;

            return {
                id: route.id,
                name: route.route_name,
                status: route.is_active ? 'Active' : 'Pending',
                orders: routeOrderCount
            };
        });

        return {
            id: staff.id,
            name: staff.name || `${staff.first_name || ''} ${staff.last_name || ''}`.trim(),
            role: staff.position || 'Staff',
            email: staff.email,
            phone: staff.phone,
            thumb_url: staff.thumb_url || null,
            active: staff.status === 'active',
            routes: formattedRoutes,
            stats: {
                completedOrders: completedOrdersCount,
                rating: 4.5 // Placeholder as rating system is not implemented yet
            }
        };
    });

    return {
        data: formattedData,
        total: result.count
    };
}

async updateStaff(id, data) {
    // Check email uniqueness if email is being updated
    if (data.email) {
        const existing = await StaffRepository.findByEmail(data.email);
        if (existing && existing.id !== parseInt(id)) {
            throw new Error('Staff with this email already exists');
        }
    }

    // Hash password if provided
    if (data.password) {
        data.password = await bcrypt.hash(data.password, 10);
    }

    // Sanitize data: convert empty strings to null
    const cleanData = { ...data };
    ['hire_date', 'thumb_url', 'position', 'phone', 'address', 'city', 'state', 'country', 'postal_code', 'notes'].forEach(field => {
        if (cleanData[field] === '') {
            cleanData[field] = null;
        }
    });

    const staff = await StaffRepository.update(id, cleanData);
    if (!staff) {
        throw new Error('Staff not found');
    }
    return staff;
}

📌 SIDE EFFECTS & WARNINGS

Operations Impact

Creating a Staff Member

  • ✓ Creates new USERS record with staff information
  • ✓ Validates email uniqueness across all users
  • ✓ Hashes password using bcrypt (10 rounds) before storage
  • ✓ Generates random password if not provided
  • ✓ Sets status to 'active' by default
  • ✓ Auto-generates created_at and updated_at timestamps
  • ✅ Creates user account that can log into the system
  • ✅ Staff can be assigned to routes, departments, and roles

Updating a Staff Member

  • ✓ Updates specified fields only (partial update supported)
  • ✓ Auto-updates updated_at timestamp
  • ✓ Validates email uniqueness if email is being changed
  • ✓ Re-hashes password if password field is provided
  • ✓ Converts empty strings to NULL for optional fields
  • ✅ May affect future payroll calculations (salary changes)
  • ✅ Department changes affect route assignments and permissions
  • ✅ Role changes affect system access and permissions

Deleting a Staff Member

  • ⚠️ Permanently removes user account from users table
  • ⚠️ Staff member will no longer be able to log in
  • ⚠️ May affect attendance records (foreign key constraints)
  • ⚠️ May affect payroll calculations if employee has pending pay
  • ⚠️ Cannot be undone (no soft delete)
  • ✅ Consider using status='inactive' or status='terminated' instead
⚠️ Important Warnings:
  • Staff and Users share the same users table
  • Deleting a staff removes their user account and login access permanently
  • Email must be unique across all users in the system
  • Salary updates will affect future payroll calculations
  • Department and role assignments affect permissions and routing
  • Attendance records reference staff via user_id FK
  • Payroll advances are tied to staff via staff_id FK
  • Use status field for deactivation instead of deleting when possible
