🏭 Production Module Documentation

Complete API documentation for Production Batches, Production Runs, Bill of Materials (BOM), and Finished Goods management

📊 Database Schema

erDiagram
    BOM ||--o{ BOMItem : contains
    BOM }o--|| Product : "produces"
    BOMItem }o--|| Product : "uses (raw material)"
    ProductionRun }o--|| Product : "produces"
    ProductionRun }o--|| BOM : "uses"

    BOM {
        int id PK
        int product_id FK
        string name
        text description
        boolean is_active
        int created_by
        datetime created_at
        datetime updated_at
    }

    BOMItem {
        int id PK
        int bom_id FK
        int product_id FK
        decimal quantity
        decimal wastage_percent
        string notes
        datetime created_at
        datetime updated_at
    }

    ProductionRun {
        int id PK
        string run_number UK
        int product_id FK
        int bom_id FK
        int quantity
        date start_date
        date end_date
        enum status
        text notes
        int created_by
        datetime created_at
        datetime updated_at
    }

    Product {
        int id PK
        string name
        string sku
        enum product_type
    }

Database Tables

Table Name Description Primary Key
boms Bill of Materials headers defining recipes for finished goods id
bom_items BOM line items containing raw materials and quantities id
production_runs Production batch/run records for manufacturing id
products Product catalog (raw materials and finished goods) id

Relationships

  • BOM → Product: Many-to-One (A BOM belongs to one finished good product)
  • BOM → BOMItem: One-to-Many (A BOM contains many items with CASCADE delete)
  • BOMItem → Product: Many-to-One (An item references one raw material product)
  • ProductionRun → Product: Many-to-One (A run produces one finished good)
  • ProductionRun → BOM: Many-to-One (Optional - A run can reference a BOM)

POST /api/production/runs

Initiate a new production run for manufacturing finished goods.

Service Logic

File: src/modules/production/production.service.js:57-77

async createRun(data, userId) {
    // Auto-generate run number
    const run_number = `RUN-${Date.now()}`;

    // Populate BOM id if not provided but product has one
    if (!data.bom_id) {
        const defaultBOM = await ProductionRepository.findOneBOMByProductId(data.product_id);
        if (defaultBOM) {
            data = { ...data, bom_id: defaultBOM.id };
        }
    }

    const runData = {
        ...data,
        run_number,
        created_by: userId,
        status: 'scheduled' // Default status
    };

    return await ProductionRepository.createRun(runData);
}

async getRuns(req, res) {
    try {
        const page = parseInt(req.query.page) || 1;
        const limit = parseInt(req.query.limit) || 10;
        const filters = {
            status: req.query.status,
            search: req.query.search
        };
        const result = await ProductionService.getAllRuns(filters, page, limit);
        return successWithPagination(res, 'Production runs retrieved successfully', result.data, {
            total: result.total,
            page,
            limit
        });
    } catch (err) {
        return error(res, err.message, 500);
    }
}

Service Logic

File: src/modules/production/production.service.js:42-49

async getAllRuns(filters = {}, page = 1, limit = 10) {
    const offset = (page - 1) * limit;
    const result = await ProductionRepository.findAllRuns(filters, limit, offset);
    return {
        data: result.rows,
        total: result.count
    };
}

Repository Logic

File: src/modules/production/production.repository.js:103-130

async findAllRuns(filters = {}, limit = 10, offset = 0) {
    const where = {};
    if (filters.status) where.status = filters.status;
    if (filters.search) {
        where[Op.or] = [
            { run_number: { [Op.like]: `%${filters.search}%` } }
        ];
    }

    return await ProductionRun.findAndCountAll({
        where,
        limit,
        offset,
        include: [
            {
                model: Product,
                as: 'product',
                attributes: ['id', 'name', 'sku']
            },
            {
                model: BOM,
                as: 'bom',
                attributes: ['id', 'name']
            }
        ],
        order: [['created_at', 'DESC']]
    });
}

File: src/modules/production/production.repository.js:74-96

async updateBOM(id, data, items) {
    const transaction = await sequelize.transaction();
    try {
        const bom = await BOM.findByPk(id);
        if (!bom) throw new Error('BOM not found');

        await bom.update(data, { transaction });

        if (items) {
            // Delete all items and re-add
            await BOMItem.destroy({ where: { bom_id: id }, transaction });
            const bomItems = items.map(item => ({ ...item, bom_id: bom.id }));
            await BOMItem.bulkCreate(bomItems, { transaction });
        }

        await transaction.commit();
        return await this.findBOMById(id);
    } catch (error) {
        await transaction.rollback();
        throw error;
    }
}

📖 References

Production Run Status Values

Status Types

  • scheduled - Run is scheduled but not started
  • in_progress - Production is active
  • completed - Production finished
  • cancelled - Run was cancelled

Status Transitions

  • scheduled → in_progress
  • in_progress → completed
  • any → cancelled (if not completed)

Product Types

  • finished_good - Manufactured products
  • raw_material - Input materials for BOMs

BOM Item Calculations

  • Base Quantity: quantity
  • With Wastage: quantity × (1 + wastage_percent/100)
  • Total for Run: per_unit × run_quantity
