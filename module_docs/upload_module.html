File: src/modules/upload/upload.controller.js uploadImage

async uploadImage(req, res) {
    try {
        // Handle both single file (req.file) and any file (req.files)
        const file = req.file || (req.files && req.files.length > 0 ? req.files[0] : null);

        if (!file) {
            return error(res, 'No file uploaded', 400);
        }

        const fileProps = this._getFileProps(file);

        // Extract metadata from request body
        const { caption, description, alt_text } = req.body;

        // Save to database
        const galleryItem = await Gallery.create({
            ...fileProps,
            caption: caption || null,
            description: description || null,
            alt_text: alt_text || null,
            uploaded_by: req.user ? req.user.id : null
        });

        return success(res, 'Image uploaded successfully', {
            ...fileProps,
            id: galleryItem.id,
            meta: { caption, description, alt_text }
        }, 201);
    } catch (err) {
        return error(res, err.message, 500);
    }
}

// Helper to get file URL and properties regardless of storage provider
_getFileProps(file) {
    if (useDO) {
        // Digital Ocean Spaces file object properties
        return {
            filename: path.basename(file.key),
            originalname: file.originalname,
            mimetype: file.mimetype,
            size: file.size,
            url: file.location,  // Full DO Space URL
            path: file.key        // Use key as path reference
        };
    } else {
        // Local file object properties
        return {
            filename: file.filename,
            originalname: file.originalname,
            mimetype: file.mimetype,
            size: file.size,
            url: `/uploads/images/${file.filename}`,
            path: file.path
        };
    }
}

Controller Logic

File: src/modules/upload/upload.controller.js uploadMultipleImages

async uploadMultipleImages(req, res) {
    try {
        if (!req.files || req.files.length === 0) {
            return error(res, 'No files uploaded', 400);
        }

        // Parse metadata JSON
        let metadata = [];
        if (req.body.metadata) {
            try { metadata = JSON.parse(req.body.metadata); }
            catch (e) { Ignore parse error }
        }

        const { caption, description, alt_text } = req.body;

        // Process each file
        const uploadPromises = req.files.map(async (file, index) => {
            const fileProps = this._getFileProps(file);

            // Determine metadata for this specific file
            let fileMeta = metadata[index] || {
                caption: Array.isArray(caption) ? caption[index] : caption,
                description: Array.isArray(description) ? description[index] : description,
                alt_text: Array.isArray(alt_text) ? alt_text[index] : alt_text
            };

            // Save to database
            const galleryItem = await Gallery.create({
                ...fileProps,
                caption: fileMeta.caption || null,
                description: fileMeta.description || null,
                alt_text: fileMeta.alt_text || null,
                uploaded_by: req.user ? req.user.id : null
            });

            return { ...fileProps, id: galleryItem.id, meta: fileMeta };
        });

        const uploadedFiles = await Promise.all(uploadPromises);

        return success(res, `${uploadedFiles.length} images uploaded successfully`, {
            count: uploadedFiles.length,
            files: uploadedFiles
        }, 201);
    } catch (err) {
        return error(res, err.message, 500);
    }
}

// File filter - only allow images
const fileFilter = (req, file, cb) => {
    const allowedTypes = ['image/jpeg', 'image/jpg', 'image/png', 'image/gif', 'image/webp'];

        cb(null, true);
    } else {
        cb(new Error('Invalid file type. Only JPEG, PNG, GIF, and WebP images are allowed.'), false);
    }
};

async listImages(req, res) {
    try {
        const page = parseInt(req.query.page, 10) || 1;
        const limit = parseInt(req.query.limit, 10) || 12;
        const offset = (page - 1) * limit;

        // Fetch from database with uploader info
        const { count, rows: images } = await Gallery.findAndCountAll({
            limit,
            offset,
            order: [['created_at', 'DESC']],
            include: [{
                association: 'uploader',
                attributes: ['id', 'name', 'email']
            }]
        });

        return successWithPagination(res, 'Images retrieved successfully', images, {
            total: count,
            page,
            limit,
            totalPage: Math.ceil(count / limit)
        });
    } catch (err) {
        return error(res, err.message, 500);
    }
}

async deleteImage(req, res) {
    try {
        const { filename } = req.params;

        // Delete from database first
        await Gallery.destroy({ where: { filename: filename } });

        if (useDO) {
            // Delete from DO Spaces
            const key = `uploads/images/${filename}`;

            try {
                await s3Client.send(new DeleteObjectCommand({
                    Bucket: process.env.DO_BUCKET,
                    Key: key
                }));
                return success(res, 'Image deleted successfully', { filename });
            } catch (s3Err) {
                console.error('S3 Delete Error:', s3Err);
                return success(res, 'Image deleted from DB but failed/missing on S3', { filename });
            }
        } else {
            // Delete from local filesystem
            const filePath = path.join(__dirname, '../../../public/uploads/images', filename);

            if (fs.existsSync(filePath)) {
                fs.unlinkSync(filePath);
            }

            return success(res, 'Image deleted successfully', { filename });
        }
    } catch (err) {
        return error(res, err.message, 500);
    }
}

async function uploadProductImage(file, metadata) {
    const formData = new FormData();
    formData.append('image', file);
    formData.append('caption', metadata.caption);
    formData.append('alt_text', metadata.alt_text);

    const response = await fetch('/api/upload/image', {
        method: 'POST',
        headers: {
            'Authorization': `Bearer ${getToken()}`
        },
        body: formData
    });

    const result = await response.json();
    return result.data.url;
}

        formData.append('metadata', metadata);

        try {
            const response = await fetch('/api/upload/images', {
                method: 'POST',
                headers: { 'Authorization': `Bearer ${token}` },
                body: formData
            });
            const result = await response.json();
            console.log(`${result.data.count} images uploaded`);
        } finally {
            setUploading(false);
        }
    };

    return (
        <div>
            <input
                type="file"
                multiple
                accept="image/*"
                onChange={(e) => setFiles([...e.target.files])}
            />
            <button onClick={handleUpload} disabled={uploading}>
                {uploading ? 'Uploading...' : 'Upload Gallery'}
            </button>
        </div>
    );
}
