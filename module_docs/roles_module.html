exports.findAll = async (page, limit) => {
  if (page && limit) {
    const offset = (page - 1) * limit;
    const { rows, count } = await Role.findAndCountAll({
      include: [RoleSettings],
      order: [['id', 'DESC']],
      limit: parseInt(limit, 10),
      offset: parseInt(offset, 10),
      distinct: true
    });

    const mapped = rows.map(r => {
      const p = r.get({ plain: true });
      const settingsRow = p.RoleSetting || p.RoleSettings;
      return {
        id: p.id,
        role: p.name,
        display_name: p.display_name,
        description: p.description,
        status: p.status,
        permissions: r.permissions,
        settings: parseSettings(settingsRow)
      };
    });
    return { rows: mapped, total: count };
  }

  // Return all if no pagination
  const rows = await Role.findAll({
    include: [RoleSettings],
    order: [['id', 'DESC']]
  });
  const mapped = rows.map(r => {
    const p = r.get({ plain: true });
    const settingsRow = p.RoleSetting || p.RoleSettings;
    return {
      id: p.id,
      role: p.name,
      display_name: p.display_name,
      description: p.description,
      status: p.status,
      permissions: r.permissions,
      settings: parseSettings(settingsRow)
    };
  });
  return { rows: mapped, total: mapped.length };
};

exports.create = async (data) => {
  // 1. Create role
  const createdRole = await Role.create({
    name: data.role || data.name,
    display_name: data.display_name,
    description: data.description,
    status: data.status || 'active',
    permissions: data.permissions || []
  });

  // 2. Create role settings
  await RoleSettings.create({
    role_id: createdRole.id,
    menu: JSON.stringify(data.menu || []),
    dashboard: JSON.stringify(data.dashboard || []),
    custom: JSON.stringify(data.custom || {}),
  });

  // 3. Fetch and return complete role
  const settings = await RoleSettings.findOne({
    where: { role_id: createdRole.id }
  });
  return {
    id: createdRole.id,
    role: createdRole.name,
    display_name: createdRole.display_name,
    description: createdRole.description,
    status: createdRole.status,
    permissions: createdRole.permissions,
    settings: parseSettings(settings)
  };
};

exports.update = async (id, data) => {
  const role = await Role.findByPk(id);
  if (!role) return null;

  // 1. Update role fields
  const roleUpdatePayload = {};
  if (data.role || data.name) roleUpdatePayload.name = data.role || data.name;
  if (data.display_name !== undefined) roleUpdatePayload.display_name = data.display_name;
  if (data.description !== undefined) roleUpdatePayload.description = data.description;
  if (data.status !== undefined) roleUpdatePayload.status = data.status;
  if (data.permissions !== undefined) roleUpdatePayload.permissions = data.permissions;

  if (Object.keys(roleUpdatePayload).length > 0) {
    await role.update(roleUpdatePayload);
  }

  // 2. Update role settings
  if (data.menu || data.dashboard || data.custom) {
    const settings = await RoleSettings.findOne({ where: { role_id: id } });
    const payload = {};
    if (data.menu !== undefined) payload.menu = JSON.stringify(data.menu || []);
    if (data.dashboard !== undefined) payload.dashboard = JSON.stringify(data.dashboard || []);
    if (data.custom !== undefined) payload.custom = JSON.stringify(data.custom || {});

    if (settings) {
      await settings.update(payload);
    } else {
      await RoleSettings.create({ role_id: id, ...payload });
    }
  }

  return exports.findById(id);
};

exports.delete = async (id) => {
  const role = await Role.findByPk(id);
  if (!role) return null;

  // SYSTEM SAFEGUARD: Prevent deleting Superadmin role
  if (role.name.toLowerCase() === 'superadmin' ||
      role.name.toLowerCase() === 'super admin') {
    throw new Error('System safeguarded: Cannot delete Superadmin role');
  }

  await Role.destroy({ where: { id } });
};

🔐 Role-Based Access Control (RBAC)

Permission System

  • ✅ Wildcard Permission - "*" grants full system access
  • ✅ Scoped Permissions - Format: "resource.action" (e.g., "sales.create", "users.delete")
  • ✅ JSON Storage - Permissions stored as JSON array in roles table

Role Status Values

  • ✅ active - Role is active and can be assigned to users
  • ⚠️ inactive - Role is disabled but not deleted

Role Settings Structure

{
  "menu": ["dashboard", "sales", "users", "reports"],
  "dashboard": ["overview", "analytics", "stats"],
  "custom": {
    "theme": "dark",
    "layout": "expanded",
    "region": "North"
  }
}

System Safeguards

  • 🛡️ Superadmin Protection - Cannot delete Superadmin role
  • 🛡️ Unique Role Names - Role names must be unique
  • 🛡️ Settings Auto-Creation - Role settings automatically created with new roles

🧪 Quick Testing

List All Roles

curl -X GET "http://localhost:5000/api/roles/list?page=1&limit=10" \
  -H "Authorization: Bearer YOUR_TOKEN"

Create Role

curl -X POST http://localhost:5000/api/roles/add \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer YOUR_TOKEN" \
  -d '{
    "role": "Sales Manager",
    "display_name": "Sales Department Manager",
    "permissions": ["sales.view", "sales.create"],
    "menu": ["sales", "dashboard"],
    "status": "active"
  }'

Update Role

curl -X PUT http://localhost:5000/api/roles/update/10 \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer YOUR_TOKEN" \
  -d '{
    "display_name": "Senior Sales Manager",
    "permissions": ["sales.view", "sales.create", "reports.view"]
  }'

Get Menu Items

curl -X GET http://localhost:5000/api/roles/menus \
  -H "Authorization: Bearer YOUR_TOKEN"

Create Menu Item

curl -X POST http://localhost:5000/api/roles/menus \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer YOUR_TOKEN" \
  -d '{
    "title": "Products",
    "path": "/products",
    "icon": "box",
    "sort_order": 3
  }'

Delete Role

curl -X DELETE http://localhost:5000/api/roles/delete/10 \
  -H "Authorization: Bearer YOUR_TOKEN"
