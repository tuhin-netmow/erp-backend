🔌 API Call Code
GET /api/sales/orders?page=1
fetchOrders.ts
const fetchOrders = async (page: number = 1) => { const token = localStorage.getItem('token'); const API_URL = import.meta.env.VITE_API_URL; const response = await fetch( `${API_URL}/api/sales/orders?page=${page}`, { headers: { 'Authorization': `Bearer ${token}` } } ); return await response.json(); };

const updateOrderStatus = async (id: number, status: string) => { const token = localStorage.getItem('token'); const API_URL = import.meta.env.VITE_API_URL; const response = await fetch( `${API_URL}/api/sales/orders/${id}/status`, { method: 'PATCH', headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ status }) } ); return await response.json(); };

const deleteOrder = async (id: number) => { const token = localStorage.getItem('token'); const API_URL = import.meta.env.VITE_API_URL; const response = await fetch( `${API_URL}/api/sales/orders/${id}`, { method: 'DELETE', headers: { 'Authorization': `Bearer ${token}` } } ); if (!response.ok) { throw new Error('Failed to delete order'); } return await response.json(); };

Order Status Flow

📊 Status Types:
  • pending - Order created, awaiting confirmation
  • confirmed - Order confirmed by sales team
  • in-transit - Order is being delivered
  • delivered - Order successfully delivered
  • cancelled - Order cancelled

Permissions

✅ Required Permissions:
  • sales.orders.view - View sales orders
  • sales.orders.update - Update order status
  • sales.orders.delete - Delete orders
