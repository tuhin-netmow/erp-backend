🔌 API Call Code (Fetch/Axios)
apiCalls.ts
// Using fetch API const fetchProducts = async (page: number = 1) => { const token = localStorage.getItem('token'); const API_URL = import.meta.env.VITE_API_URL; try { const response = await fetch( `${API_URL}/api/products?page=${page}&limit=10`, { method: 'GET', headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' } } ); if (!response.ok) { throw new Error('Failed to fetch products'); } const data = await response.json(); setProducts(data.data); setTotalPages(data.pagination.totalPages); } catch (error) { console.error('Error fetching products:', error); } }; // Using Axios import axios from 'axios'; const fetchProductsAxios = async (page: number = 1) => { try { const response = await axios.get( `${API_URL}/api/products`, { params: { page: page, limit: 10 }, headers: { 'Authorization': `Bearer ${token}` } } ); setProducts(response.data.data); setTotalPages(response.data.pagination.totalPages); } catch (error) { console.error('Error fetching products:', error); } };

const searchProducts = async (query: string) => { const response = await fetch( `${API_URL}/api/products?search=${encodeURIComponent(query)}`, { headers: { 'Authorization': `Bearer ${token}` } } ); const data = await response.json(); return data.data; }; // Debounced search import { useDebounce } from 'use-debounce'; const debouncedSearch = useDebounce((query) => { if (query) { searchProducts(query); } }, 500);
