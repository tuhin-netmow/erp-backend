🛒 Sales Module

Complete sales order management system with invoicing, payments, delivery tracking, warehouses, and sales routes

📋 API Routes

📌 Orders Management

POST /api/sales/orders

Create a new sales order with line items. Auto-generates order_number and total_amount with stock deduction.

GET /api/sales/orders

Get all orders with pagination. Supports filtering by status, customer, date range, and search by order number.

GET /api/sales/orders/:id

Get detailed order information by ID with customer, items, products, and delivery details.

PUT /api/sales/orders/:id

Update order details and status. Returns complete order data.

DELETE /api/sales/orders/:id

Cancel/delete a sales order.

📌 Invoices & Payments

GET /api/sales/invoices

Get all invoices with pagination.

POST /api/sales/orders/:id/invoice

Create an invoice for a specific order.

POST /api/sales/orders/:id/pay

Record a payment for an order (supports negative amounts for refunds).

📌 Deliveries

POST /api/sales/orders/:id/deliver

Create a delivery record and update order status.

PUT /api/sales/deliveries/:id

Update delivery status and tracking information.

async createOrder(data, userId) {
  let { items, sales_tax_percent, ...orderInfo } = data;

  // Generate a unique order number using prefix settings
  let order_number = data.order_number;
  if (!order_number) {
    try {
      const { generateNumber } = require('../settings/numbering.helper');
      order_number = await generateNumber('order');
    } catch (err) {
      console.error('Error generating order number:', err);
      order_number = `ORD-${Date.now()}`;
    }
  }

  // Calculate total amount, discount amount, and tax amount
  let total_amount = 0;
  let discount_amount = 0;
  let item_level_tax = 0; // Tax from individual products

  if (items && Array.isArray(items)) {
    const { ProductRepository } = require('../products/products.repository');

    // Use Promise.all to handle async product fetching
    items = await Promise.all(items.map(async (item) => {
      const quantity = Number(item.quantity) || 0;
      const unit_price = Number(item.unit_price) || 0;
      const discount = Number(item.discount) || 0;

      // Fetch product to get default sales_tax
      const product = await ProductRepository.findById(item.product_id);
      // Use provided tax rate from frontend (e.g. POS) or fallback to product's default
      const sales_tax_rate = item.sales_tax !== undefined ? Number(item.sales_tax) :
        (product && product.sales_tax ? Number(product.sales_tax) : 0);

      const total_price = quantity * unit_price; // Before discount
      const line_total = total_price - discount; // After discount

      // Calculate item tax (from product's sales_tax) on line_total
      const item_tax_amount = (line_total * sales_tax_rate) / 100;

      discount_amount += discount;
      total_amount += total_price; // Sum of total_price (before discount)
      item_level_tax += item_tax_amount;

      return {
        ...item,
        tax_amount: item_tax_amount,
        sales_tax_percent: sales_tax_rate,  // Store the tax percentage
        purchase_cost: product ? product.cost : 0 // Lock in the cost at time of sale
      };
    }));
  }

  // Calculate order-level tax if sales_tax_percent is provided
  let order_level_tax = 0;
  const tax_percent = sales_tax_percent ? Number(sales_tax_percent) : 0;

  if (tax_percent > 0) {
    // Apply order-level tax on total_amount (after item discounts)
    order_level_tax = (total_amount * tax_percent) / 100;
  }

  // Total tax is sum of item-level tax and order-level tax
  const tax_amount = item_level_tax + order_level_tax;

  const orderData = {
    ...orderInfo,
    order_number,
    total_amount,
    discount_amount,
    tax_amount,
    sales_tax_percent: tax_percent,
    status: 'pending', // Force status to pending (dummy order)
    created_by: userId
  };

  const order = await OrderRepository.create(orderData, items || []);

  // Record initial status in history
  await OrderStatusHistoryRepository.create({
    order_id: order.id,
    status: order.status || 'pending',
    changed_by: userId,
    notes: 'Initial order creation'
  });

  // Deduct stock and record movements for each item
  const StockMovement = require('../products/stock-movement.model');
  const { ProductRepository: ProductRepo } = require('../products/products.repository');

  if (items && Array.isArray(items)) {
    for (const item of items) {
      // Get current product stock
      const product = await ProductRepo.findById(item.product_id);
      if (!product) {
        throw new Error(`Product with ID ${item.product_id} not found`);
      }

      // Check if sufficient stock
      if (product.stock_quantity < item.quantity) {
        throw new Error(`Insufficient stock for product ${product.name}. Available: ${product.stock_quantity}, Required: ${item.quantity}`);
      }

      // Deduct stock
      const newStock = product.stock_quantity - item.quantity;
      await ProductRepo.update(item.product_id, { stock_quantity: newStock });

      // Record stock movement
      await StockMovement.create({
        product_id: item.product_id,
        movement_type: 'sale',
        quantity: -item.quantity, // Negative for stock out
        reference_type: 'order',
        reference_id: order.id,
        notes: `Stock deducted for order ${order_number}`,
        created_by: userId
      });
    }
  }

  return order;
}

File: src/modules/sales/sales.repository.js OrderRepository.create

async create(orderData, itemsData) {
  const transaction = await sequelize.transaction();
  try {
    const order = await Order.create(orderData, { transaction });

    const items = itemsData.map(item => {
      const discount = item.discount || 0;
      const total_price = item.quantity * item.unit_price; // Before discount
      const line_total = item.line_total || (total_price - discount); // After discount

      return {
        ...item,
        order_id: order.id,
        discount: discount,
        total_price: total_price,
        line_total: line_total,
        tax_amount: item.tax_amount || 0,
        sales_tax_percent: item.sales_tax_percent || 0
      };
    });

    if (items.length > 0) {
      await OrderItem.bulkCreate(items, { transaction });
    }

    await transaction.commit();
    return await this.findById(order.id);
  } catch (error) {
    await transaction.rollback();
    throw error;
  }
}

async getAllOrders(filters = {}, page = 1, limit = 10) {
  const offset = (page - 1) * limit;
  const result = await OrderRepository.findAll(filters, limit, offset);

  // Transform deliveries array to single delivery object
  const transformedRows = result.rows.map(order => {
    const orderData = order.toJSON();

    // Convert deliveries array to single delivery object (latest one)
    orderData.delivery = orderData.deliveries && orderData.deliveries.length > 0
      ? orderData.deliveries[0]
      : null;

    // Add delivery_status field
    orderData.delivery_status = orderData.delivery ? orderData.delivery.status : null;

    delete orderData.deliveries;

    // Calculate Payment Totals
    const payments = orderData.payments || [];
    const grossPaid = payments.reduce((sum, p) => {
      const amt = parseFloat(p.amount);
      return amt > 0 ? sum + amt : sum;
    }, 0);
    const refunded = payments.reduce((sum, p) => {
      const amt = parseFloat(p.amount);
      return amt < 0 ? sum + Math.abs(amt) : sum;
    }, 0);
    const netPaid = grossPaid - refunded;

    orderData.gross_paid_amount = grossPaid;
    orderData.refunded_amount = refunded;
    orderData.total_paid_amount = netPaid;
    orderData.total_payable_amount = parseFloat(orderData.total_amount)
      - parseFloat(orderData.discount_amount || 0)
      + parseFloat(orderData.tax_amount || 0);

    return orderData;
  });

  return {
    data: transformedRows,
    total: result.count
  };
}

File: src/modules/sales/sales.repository.js OrderRepository.findAll

async findAll(filters = {}, limit = 10, offset = 0) {
  const where = {};
  if (filters.status) {
    if (filters.status === 'in_transit') {
      where.status = { [Op.or]: ['in_transit', 'shipped'] };
    } else {
      where.status = filters.status;
    }
  }
  if (filters.customer_id) where.customer_id = filters.customer_id;

  if (filters.search) {
    where[Op.or] = [
      { order_number: { [Op.like]: `%${filters.search}%` } }
    ];
  }

  return await Order.findAndCountAll({
    where,
    limit,
    offset,
    include: [
      {
        model: Customer,
        as: 'customer',
        attributes: ['id', 'name', 'email', 'phone', 'company']
      },
      {
        model: OrderItem,
        as: 'items',
        include: [
          {
            model: Product,
            as: 'product',
            attributes: ['id', 'name', 'sku', 'price', 'image_url', 'specification']
          }
        ]
      },
      {
        model: Delivery,
        as: 'deliveries',
        limit: 1,
        order: [['created_at', 'DESC']],
        separate: true
      },
      {
        model: Payment,
        as: 'payments',
        separate: true
      }
    ],
    order: [['created_at', 'DESC']],
    distinct: true
  });
}

-- Query orders with relationships
SELECT
  o.id, o.order_number, o.customer_id, o.order_date,
  o.status, o.total_amount, o.tax_amount, o.discount_amount,
  o.shipping_address, o.billing_address, o.payment_status,
  o.notes, o.due_date, o.delivery_date, o.created_at,
  c.id as customer_id, c.name as customer_name, c.email, c.phone,
  p.id as payment_id, p.amount, p.payment_method, p.reference_number,
  d.id as delivery_id, d.delivery_number, d.delivery_date, d.status as delivery_status
FROM orders o
LEFT JOIN customers c ON o.customer_id = c.id
LEFT JOIN payments p ON o.id = p.order_id
LEFT JOIN deliveries d ON o.id = d.order_id
WHERE o.status = ?
ORDER BY o.created_at DESC
LIMIT 10 OFFSET 0;

🧪 Quick Testing

Create Order

curl -X POST http://localhost:5000/api/sales/orders \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer YOUR_TOKEN" \
  -d '{
    "customer_id": 1,
    "shipping_address": "123 Main St, New York, NY",
    "items": [
      {
        "product_id": 1,
        "quantity": 2,
        "unit_price": 50.00,
        "line_total": 100.00
      }
    ]
  }'

Get All Orders

curl -X GET "http://localhost:5000/api/sales/orders?page=1&limit=10&status=pending" \
  -H "Authorization: Bearer YOUR_TOKEN"

Record Payment

curl -X POST http://localhost:5000/api/sales/orders/1/pay \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer YOUR_TOKEN" \
  -d '{
    "amount": 150.00,
    "payment_method": "cash",
    "reference_number": "REF-123456"
  }'
