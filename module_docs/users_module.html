👥 Users Module

Complete user account management system with CRUD operations, role-based access control, and department integration

📋 API Routes

GET /api/users

Get all users with pagination. Supports filtering by role, department, status, and text search.

POST /api/users/add

Create a new user account with automatic password hashing and email validation.

GET /api/users/get/:id

Get detailed user information by ID with role and department data.

PUT /api/users/update/:id

Update user information. Supports partial updates and password rehashing.

DELETE /api/users/delete/:id

Delete a user account (soft delete supported).

File: src/modules/users/user.repository.js findAll

async findAll(filters = {}, limit = 10, offset = 0) {
  const where = {};

  // Build filters
  if (filters.role_id) where.role_id = filters.role_id;
  if (filters.department_id) where.department_id = filters.department_id;
  if (filters.status) where.status = filters.status;

  // Search across multiple fields
  if (filters.search) {
    where[Op.or] = [
      { name: { [Op.like]: `%${filters.search}%` } },
      { first_name: { [Op.like]: `%${filters.search}%` } },
      { last_name: { [Op.like]: `%${filters.search}%` } },
      { email: { [Op.like]: `%${filters.search}%` } }
    ];
  }

  return await User.findAndCountAll({
    where,
    limit,
    offset,
    include: [
      { model: Role, as: 'role', attributes: ['id', 'name'] },
      { model: Department, as: 'department', attributes: ['id', 'name'] }
    ],
    attributes: { exclude: ['password'] },
    order: [['created_at', 'DESC']]
  });
}

  d.id as department_id, d.name as department_name
FROM users u
LEFT JOIN roles r ON u.role_id = r.id
LEFT JOIN departments d ON u.department_id = d.id
WHERE u.status = 'active'
ORDER BY u.created_at DESC

async createUser(data, userId) {
  // 1. Check if email already exists
  const existing = await UserRepository.findByEmail(data.email);
  if (existing) {
    throw new Error('User with this email already exists');
  }

  // 2. Hash password with bcrypt (10 salt rounds)
  const hashedPassword = await bcrypt.hash(data.password, 10);
  const userData = { ...data, password: hashedPassword, created_by: userId };

  // 3. Create user
  const user = await UserRepository.create(userData);
  const { password, ...userWithoutPassword } = user.toJSON();
  return userWithoutPassword;
}

File: src/modules/users/user.service.js updateUser

async updateUser(id, data) {
  // 1. If email is being updated, check for duplicates
  if (data.email) {
    const existing = await UserRepository.findByEmail(data.email);
    if (existing && existing.id !== parseInt(id)) {
      throw new Error('User with this email already exists');
    }
  }

  // 2. Hash password if being updated
  if (data.password) {
    data.password = await bcrypt.hash(data.password, 10);
  }

  // 3. Update user
  const user = await UserRepository.update(id, data);
  if (!user) {
    throw new Error('User not found');
  }
  const { password, ...userWithoutPassword } = user.toJSON();
  return userWithoutPassword;
}

🔐 User Status Flow

Status Values

  • ✅ active - User is active and can access the system
  • ⚠️ inactive - User account is temporarily disabled
  • ❌ terminated - User employment has ended
  • 🏖️ on_leave - User is currently on leave

Security Features

  • ✅ Password Hashing - All passwords hashed using bcrypt (10 rounds)
  • ✅ Email Uniqueness - Email addresses must be unique across all users
  • ✅ Password Exclusion - Password never returned in API responses
  • ✅ Soft Delete - Users can be soft deleted with deleted_at timestamp
  • ✅ JWT Authentication - All endpoints protected with verifyToken middleware
