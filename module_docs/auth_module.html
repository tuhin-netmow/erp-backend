🔐 Auth Module

Complete authentication and authorization system with JWT tokens, role-based access control, and user management

📋 API Routes

POST /api/auth/login

User login with email and password. Returns JWT token, user info, menus, and dashboards.

POST /api/auth/register

Create a new user account. Returns user data and JWT token.

GET /api/auth/me

Get current authenticated user details with role, menus, and dashboards. Requires JWT token.

POST /api/auth/refresh

Refresh an expired JWT token using a valid refresh token.

POST /api/auth/logout

Logout current user. Requires JWT token.

PUT /api/auth/profile

Update current user profile information. Requires JWT token.

File: src/modules/auth/auth.service.js exports.login

exports.login = async (email, password) => {
  // 1. Find user by email (with role)
  const user = await repo.findByEmailWithRole(email);
  if (!user) throw { status: 400, message: 'Invalid credentials' };

  // 2. Compare password using bcrypt
  const ok = await bcrypt.compare(password, user.password);
  if (!ok) throw { status: 400, message: 'Invalid credentials' };

  // 3. Generate JWT token
  const token = jwt.sign(
    { id: user.id, email: user.email, role_id: user.role_id },
    jwtSecret,
    { expiresIn: jwtExpiresIn }
  );

  // 4. Get role menus and dashboards
  const roleMenus = await repo.getRoleMenus();
  const roleDashboards = await repo.getRoleDashboards();

  // 5. Build hierarchical menu structure
  const buildMenuTree = (menus, parentId = null) => {
    return menus
      .filter(menu => menu.parent_id === parentId)
      .map(menu => ({
        id: menu.id,
        title: menu.title,
        icon: menu.icon,
        path: menu.path,
        children: buildMenuTree(menus, menu.id)
      }));
  };

  const menuTree = buildMenuTree(roleMenus);

  // 6. Remove password from response
  const userData = user.toJSON();
  delete userData.password;

  // 7. Return user, token, menus, dashboards
  return {
    user: userData,
    token,
    menus: menuTree,
    dashboards: roleDashboards
  };
};

exports.register = async (userData) => {
  // 1. Check if user already exists
  const existing = await repo.findByEmail(userData.email);
  if (existing) throw { status: 400, message: 'Email already registered' };

  // 2. Hash password with bcrypt (10 salt rounds)
  const hashedPassword = await bcrypt.hash(userData.password, 10);
  userData.password = hashedPassword;

  // 3. Create user
  const user = await repo.create(userData);
  delete user.password;

  // 4. Generate JWT token
  const token = jwt.sign(
    { id: user.id, email: user.email, role_id: user.role_id },
    jwtSecret,
    { expiresIn: jwtExpiresIn }
  );

  return { user, token };
};

exports.getCurrentUser = async (userId) => {
  // 1. Find user with role
  const user = await repo.findByIdWithRole(userId);
  if (!user) throw { status: 404, message: 'User not found' };

  // 2. Get role menus and dashboards
  const roleMenus = await repo.getRoleMenus();
  const roleDashboards = await repo.getRoleDashboards();

  // 3. Build hierarchical menu structure
  const buildMenuTree = (menus, parentId = null) => {
    return menus
      .filter(menu => menu.parent_id === parentId)
      .map(menu => ({
        id: menu.id,
        title: menu.title,
        icon: menu.icon,
        path: menu.path,
        is_active: menu.is_active,
        children: buildMenuTree(menus, menu.id)
      }));
  };

  const menuTree = buildMenuTree(roleMenus);

  // 4. Remove password from response
  const userData = user.toJSON();
  delete userData.password;
  userData.phone = userData.phone || null;
  userData.profile_image = userData.thumb_url || null;

  // 5. Return complete profile
  return {
    user: userData,
    menus: menuTree,
    dashboards: roleDashboards
  };
};

exports.refresh = async (req, res, next) => {
  try {
    const { refreshToken, refresh_token } = req.body;
    const token = refreshToken || refresh_token;

    if (!token) {
      return res.status(400).json({
        status: false,
        message: 'Validation Error',
        errors: [{ path: ['refreshToken'], message: 'Refresh token is required' }]
      });
    }

    const result = await service.refreshToken(token);
    return success(res, 'Token refreshed successfully', result);
  } catch (err) { next(err); }
};

Service Logic

File: src/modules/auth/auth.service.js exports.refreshToken

exports.refreshToken = async (token) => {
  try {
    // 1. Verify the refresh token
    const decoded = jwt.verify(token, jwtSecret);

    // 2. Find user
    const user = await repo.findById(decoded.id);
    if (!user) throw { status: 404, message: 'User not found' };

    // 3. Generate new access token
    const newToken = jwt.sign(
      { id: user.id, email: user.email, role_id: user.role_id },
      jwtSecret,
      { expiresIn: jwtExpiresIn }
    );

    return { token: newToken };
  } catch (err) {
    throw { status: 401, message: 'Invalid or expired refresh token' };
  }
};

exports.updateProfile = async (userId, data) => {
  // 1. Prevent updating sensitive fields
  delete data.password;
  delete data.role_id;
  delete data.email;

  // 2. Map profile_image to thumb_url
  if (data.profile_image) {
    data.thumb_url = data.profile_image;
  }

  // 3. Update user
  const updatedUser = await repo.update(userId, data);
  if (!updatedUser) throw { status: 404, message: 'User not found' };

  // 4. Remove password from response
  const userData = updatedUser.toJSON();
  delete userData.password;

  // 5. Ensure fields are present in response
  userData.phone = userData.phone || null;
  userData.profile_image = userData.thumb_url || userData.profile_image || null;

  return userData;
};

exports.update = async (id, data) => {
  // Try to find and update in User table first

  }

  // If not found, try Staff table

  if (staff) {
    return await staff.update(data);
  }

  return null;

Token-Based Authentication

  • ✅ JWT (JSON Web Token) for stateless authentication
  • ✅ Access Token: Short-lived (1 day default) for API requests
  • ✅ Refresh Token: Long-lived token for obtaining new access tokens
  • ✅ Middleware: verifyToken() checks Authorization header on protected routes

Role-Based Access Control (RBAC)

  • ✅ Each user assigned to one Role
  • ✅ Roles have Permissions array (JSON format)
  • ✅ Roles have custom Menus and Dashboards
  • ✅ Role Settings store UI configurations per role

Permission Examples

{
  "permissions": [
    "users.create",
    "users.read",
    "users.update",
    "users.delete",
    "sales.create",
    "sales.read",
    "inventory.manage",
    "reports.view"
  ]
}

🧪 Quick Testing

Login

curl -X POST http://localhost:5000/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{
    "email": "admin@example.com",
    "password": "SecurePassword123!"
  }'

Get Current User

curl -X GET http://localhost:5000/api/auth/me \
  -H "Authorization: Bearer YOUR_TOKEN"

Refresh Token

curl -X POST http://localhost:5000/api/auth/refresh \
  -H "Content-Type: application/json" \
  -d '{
    "refreshToken": "YOUR_REFRESH_TOKEN"
  }'

Logout

curl -X POST http://localhost:5000/api/auth/logout \
  -H "Authorization: Bearer YOUR_TOKEN"
